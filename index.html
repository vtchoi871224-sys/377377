노트북 장비 정보 수집 GUI

PowerShell · 장비 정보 표시 및 K 키 종료 
Add-Type -AssemblyName System.Windows.Forms
Add-Type -AssemblyName System.Drawing

# 장비 정보 수집
$bios = Get-CimInstance Win32_BIOS
$ip = Get-NetIPAddress -AddressFamily IPv4 | Where-Object { $_.InterfaceAlias -notmatch 'Loopback' -and $_.IPAddress -notlike '169.254*' } | Select-Object -First 1
$nic = Get-NetAdapter | Where-Object Status -eq 'Up' | Select-Object -First 1

# 날짜 및 네트워크 정보
$date = if ($bios.ReleaseDate) { $bios.ReleaseDate.ToString('yyyy-MM-dd') } else { '확인 불가' }
$ipaddr = if ($ip) { $ip.IPAddress } else { '확인 불가' }
$mac = if ($nic) { $nic.MacAddress -replace '-', ':' } else { '확인 불가' }

# GUI 생성
$form = New-Object System.Windows.Forms.Form
$form.Text = '노트북 장비 정보'
$form.Size = New-Object System.Drawing.Size(500,440)
$form.StartPosition = 'CenterScreen'
$form.BackColor = [System.Drawing.Color]::White

# 표시 항목 등 기타 코드는 이전과 동일
[void]$form.ShowDialog()
